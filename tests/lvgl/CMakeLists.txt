cmake_minimum_required(VERSION 3.16)
project(ui_keyboard_test LANGUAGES C CXX ASM)
if(NOT LVGL_SOURCE_DIR)
    message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the LVGL v9.5.0 checkout")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(ui_keyboard_test ui_keyboard_test.cpp)
target_compile_features(ui_keyboard_test PRIVATE cxx_std_17)
target_compile_options(ui_keyboard_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_include_directories(ui_keyboard_test PRIVATE ../../src)
target_link_libraries(ui_keyboard_test PRIVATE lvgl::lvgl m)
enable_testing()
add_test(NAME ui_keyboard COMMAND ui_keyboard_test)
